function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // --- A. ログイン認証 (action=login) --- if (action === 'login') { const rawName = String(e.parameter.name || e.parameter.id || ''); const rawPin = String(e.parameter.pin || ''); // 全角・半角スペースを全て削除して比較用に小文字化 const cleanInputName = rawName.replace(/[\s\u3000]/g, '').toLowerCase(); const cleanInputPin = rawPin.trim(); const sheet = ss.getSheetByName('Users'); if (!sheet) return createJsonResponse({ success: false, message: 'Usersシートが見つかりません' }); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return createJsonResponse({ success: false, message: 'Usersシートにデータがありません' }); const headers = data[0].map(h => String(h).trim()); // 柔軟な列判定 let nameIdx = headers.findIndex(h => h.includes('名') || h.includes('ユーザー') || h.includes('ID')); let pinIdx = headers.findIndex(h => h.toUpperCase().includes('PIN') || h.includes('暗証番号') || h.includes('パスワード')); let ptIdx = headers.findIndex(h => h.includes('ポイント') || h.includes('pt') || h.includes('PT')); // デフォルト位置のフォールバック (1列目:ID/名前, 2列目:名前/PIN...) if (nameIdx === -1) nameIdx = 0; if (pinIdx === -1) pinIdx = 2; // 想定: ID(0), カタカナ名(1), PIN(2) let userFound = false; let pinMatched = false; let foundUser = null; for (let i = 1; i < data.length; i++) { const row = data[i]; // 1列目と2列目の両方を名前候補としてチェック const col0 = String(row[0] || '').replace(/[\s\u3000]/g, '').toLowerCase(); const col1 = String(row[1] || '').replace(/[\s\u3000]/g, '').toLowerCase(); const isNameMatch = (col0 && col0 === cleanInputName) || (col1 && col1 === cleanInputName); if (isNameMatch) { userFound = true; // PINの比較 (数値変換、4桁ゼロ埋め変換などを全て試す) const dbPinRaw = String(row[pinIdx] !== undefined ? row[pinIdx] : row[2] || '').trim(); const dbPinPadded = dbPinRaw.padStart(4, '0'); // "826" -> "0826" if (dbPinRaw === cleanInputPin || dbPinPadded === cleanInputPin || Number(dbPinRaw) === Number(cleanInputPin)) { pinMatched = true; foundUser = { name: row[nameIdx] || row[1] || row[0], points: ptIdx !== -1 ? row[ptIdx] : (row[3] || 0) }; break; } } } if (foundUser) { return createJsonResponse({ success: true, name: String(foundUser.name), points: Number(foundUser.points) }); } else if (userFound && !pinMatched) { return createJsonResponse({ success: false, message: 'PIN（暗証番号）が一致しません' }); } else { return createJsonResponse({ success: false, message: '「' + rawName + '」というお名前が見つかりません。スプレッドシートの表記をご確認ください' }); } } // --- B. ランキング取得 (action=ranking) --- if (action === 'ranking') { const sheet = ss.getSheetByName('Users'); if (!sheet) return createJsonResponse([]); const data = sheet.getDataRange().getValues(); const headers = data[0]; const result = []; for (let i = 1; i < data.length; i++) { let obj = {}; headers.forEach((h, idx) => { obj[h] = data[i][idx]; }); result.push(obj); } return createJsonResponse(result); } // --- C. レース一覧取得 (デフォルト) --- const racesSheet = ss.getSheetByName('Races'); if (!racesSheet) return createJsonResponse([]); const racesData = racesSheet.getDataRange().getValues(); const racesHeaders = racesData[0]; const racesList = []; for (let i = 1; i < racesData.length; i++) { let obj = {}; racesHeaders.forEach((h, idx) => { obj[h] = racesData[i][idx]; }); racesList.push(obj); } return createJsonResponse(racesList); } function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (err) { return createJsonResponse({ success: false, message: '混雑しています。再度お試しください。' }); } try { const contents = JSON.parse(e.postData.contents); const rawName = String(contents.userName || contents.userId || ''); const rawPin = String(contents.pin || ''); const raceId = String(contents.raceId || '').trim(); const betAthlete = String(contents.betAthlete || '').trim(); const betPt = Number(contents.betPt || 0); const cleanInputName = rawName.replace(/[\s\u3000]/g, '').toLowerCase(); const cleanInputPin = rawPin.trim(); if (!cleanInputName || !cleanInputPin || !raceId || !betAthlete || betPt <= 0) { return createJsonResponse({ success: false, message: '入力内容が不正です' }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); const usersSheet = ss.getSheetByName('Users'); const data = usersSheet.getDataRange().getValues(); const headers = data[0].map(h => String(h).trim()); let ptIdx = headers.findIndex(h => h.includes('ポイント') || h.includes('pt') || h.includes('PT')); if (ptIdx === -1) ptIdx = 3; let userRowIndex = -1; let currentPt = 0; let canonicalName = rawName; for (let i = 1; i < data.length; i++) { const row = data[i]; const col0 = String(row[0] || '').replace(/[\s\u3000]/g, '').toLowerCase(); const col1 = String(row[1] || '').replace(/[\s\u3000]/g, '').toLowerCase(); const dbPinRaw = String(row[2] || '').trim(); const dbPinPadded = dbPinRaw.padStart(4, '0'); const isNameMatch = (col0 && col0 === cleanInputName) || (col1 && col1 === cleanInputName); const isPinMatch = (dbPinRaw === cleanInputPin || dbPinPadded === cleanInputPin || Number(dbPinRaw) === Number(cleanInputPin)); if (isNameMatch && isPinMatch) { userRowIndex = i + 1; currentPt = Number(row[ptIdx] || 0); canonicalName = row[1] || row[0] || rawName; break; } } if (userRowIndex === -1) { return createJsonResponse({ success: false, message: 'ユーザー認証に失敗しました' }); } if (currentPt < betPt) { return createJsonResponse({ success: false, message: '所持ポイントが不足しています' }); } // 1. ポイント減算 const newPt = currentPt - betPt; usersSheet.getRange(userRowIndex, ptIdx + 1).setValue(newPt); // 2. Betsシート記録 let betsSheet = ss.getSheetByName('Bets'); if (!betsSheet) { betsSheet = ss.insertSheet('Bets'); betsSheet.appendRow(['日時', 'カタカナ名', 'レースID', '賭けた選手', '賭けたポイント']); } betsSheet.appendRow([new Date(), canonicalName, raceId, betAthlete, betPt]); return createJsonResponse({ success: true, newPt: newPt }); } catch (err) { return createJsonResponse({ success: false, message: 'エラーが発生しました: ' + err.toString() }); } finally { lock.releaseLock(); } } function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
